cmake_minimum_required(VERSION 3.16)
project(antiquant_k128 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(name CANN_ROOT OPP_ROOT NATIVE_MM_HEADER COMMON_INCLUDE)
  if(NOT ${name})
    message(FATAL_ERROR "Missing ${name}")
  endif()
endforeach()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Requires Linux aarch64 CANN 9.0.0 receiver")
endif()
set(api "${OPP_ROOT}/vendors/customize/op_api")
foreach(name ascendcl opapi nnopbase acl_op_compiler)
  find_library(LIB_${name} NAMES ${name} PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
  if(NOT LIB_${name})
    message(FATAL_ERROR "Missing ${name}")
  endif()
endforeach()
find_library(LIB_custom NAMES cust_opapi PATHS "${api}/lib" NO_DEFAULT_PATH)
if(NOT LIB_custom OR NOT EXISTS "${api}/include/aclnn_d_flash_anti_quant_k128.h")
  message(FATAL_ERROR "Missing independent AntiQuant ACLNN library/header")
endif()
foreach(mode native custom)
  add_executable(antiquant_${mode} runner.cpp)
  target_include_directories(antiquant_${mode} PRIVATE "${CANN_ROOT}/include" "${COMMON_INCLUDE}")
  target_compile_options(antiquant_${mode} PRIVATE -O2)
  target_link_libraries(antiquant_${mode} PRIVATE "${LIB_ascendcl}" "${LIB_nnopbase}")
endforeach()
target_compile_definitions(antiquant_native PRIVATE ANTIQUANT_NATIVE_REFERENCE=1 ANTIQUANT_MM_HEADER="${NATIVE_MM_HEADER}")
target_link_libraries(antiquant_native PRIVATE "${LIB_opapi}")
target_include_directories(antiquant_custom PRIVATE "${api}/include")
target_link_libraries(antiquant_custom PRIVATE "${LIB_custom}" "${LIB_acl_op_compiler}")
