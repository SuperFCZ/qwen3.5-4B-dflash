cmake_minimum_required(VERSION 3.16)
project(c16_diagnostic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_library(ACL_LIBRARY ascendcl PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64" "${CANN_ROOT}/x86_64-linux/lib64" NO_DEFAULT_PATH REQUIRED)
add_executable(c16_diagnostic_runner ../fp16_dynamic_runner.cpp)
target_compile_definitions(c16_diagnostic_runner PRIVATE DFLASH_C16_DIAGNOSTIC_ONLY=1)
target_include_directories(c16_diagnostic_runner PRIVATE "${CANN_ROOT}/include")
target_link_libraries(c16_diagnostic_runner PRIVATE "${ACL_LIBRARY}")
