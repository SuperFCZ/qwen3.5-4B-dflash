cmake_minimum_required(VERSION 3.16)
project(dflash_head_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CANN_ROOT)
  message(FATAL_ERROR "Pass -DCANN_ROOT")
endif()
set(cann_paths "${CANN_ROOT}/lib64" "${CANN_ROOT}/x86_64-linux/lib64" "${CANN_ROOT}/aarch64-linux/lib64")
find_library(ASCENDCL_LIBRARY ascendcl PATHS ${cann_paths} NO_DEFAULT_PATH)
if(NOT ASCENDCL_LIBRARY)
  message(FATAL_ERROR "ascendcl missing from active CANN")
endif()
add_executable(head_native_test runner.cpp)
target_compile_definitions(head_native_test PRIVATE HEAD_NATIVE_ONLY=1)
target_include_directories(head_native_test PRIVATE "${CANN_ROOT}/include")
target_link_libraries(head_native_test PRIVATE "${ASCENDCL_LIBRARY}")
if(ASCEND_OPP_PATH)
  set(vendor "${ASCEND_OPP_PATH}/vendors/customize/op_api")
  find_library(CUST_LIBRARY cust_opapi PATHS "${vendor}/lib" NO_DEFAULT_PATH)
  find_library(COMPILER_LIBRARY acl_op_compiler PATHS ${cann_paths} NO_DEFAULT_PATH)
  find_library(NNOP_LIBRARY nnopbase PATHS ${cann_paths} NO_DEFAULT_PATH)
  foreach(library CUST_LIBRARY COMPILER_LIBRARY NNOP_LIBRARY)
    if(NOT ${library})
      message(FATAL_ERROR "${library} missing from active CANN/isolated OPP")
    endif()
  endforeach()
  add_executable(head_custom_test runner.cpp)
  target_include_directories(head_custom_test PRIVATE "${CANN_ROOT}/include" "${vendor}/include")
  target_link_libraries(head_custom_test PRIVATE "${ASCENDCL_LIBRARY}" "${CUST_LIBRARY}" "${COMPILER_LIBRARY}" "${NNOP_LIBRARY}")
endif()
