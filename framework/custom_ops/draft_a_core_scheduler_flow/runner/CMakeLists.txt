cmake_minimum_required(VERSION 3.16)
project(projection_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(projection_runner main.cpp)
target_include_directories(projection_runner PRIVATE "${COMMON_INCLUDE}" "${PRIVATE_OPP}/vendors/customize/op_api/include" "${CANN_ROOT}/include")
foreach(lib ascendcl nnopbase acl_op_compiler cust_opapi)
  if(lib STREQUAL "cust_opapi")
    set(search "${PRIVATE_OPP}/vendors/customize/op_api/lib")
  else()
    set(search "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64")
  endif()
  find_library(LIB_${lib} ${lib} PATHS ${search} NO_DEFAULT_PATH)
  if(NOT LIB_${lib})
    message(FATAL_ERROR "Missing bound library: ${lib}")
  endif()
  target_link_libraries(projection_runner PRIVATE "${LIB_${lib}}")
endforeach()
