cmake_minimum_required(VERSION 3.16)
project(smoke_add_test LANGUAGES CXX)

if(NOT CANN_ROOT OR NOT ASCEND_OPP_PATH)
    message(FATAL_ERROR "Pass -DCANN_ROOT and -DASCEND_OPP_PATH")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(vendor_api "${ASCEND_OPP_PATH}/vendors/customize/op_api")
add_executable(smoke_add_test main.cpp)
target_include_directories(smoke_add_test PRIVATE "${CANN_ROOT}/include" "${vendor_api}/include")
find_library(ASCENDCL_LIBRARY ascendcl PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/x86_64-linux/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
find_library(CUST_OPAPI_LIBRARY cust_opapi PATHS "${vendor_api}/lib" NO_DEFAULT_PATH)
find_library(ACL_OP_COMPILER_LIBRARY acl_op_compiler PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/x86_64-linux/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
find_library(NNOPBASE_LIBRARY nnopbase PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/x86_64-linux/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
foreach(library ASCENDCL_LIBRARY CUST_OPAPI_LIBRARY ACL_OP_COMPILER_LIBRARY NNOPBASE_LIBRARY)
    if(NOT ${library})
        message(FATAL_ERROR "Required library ${library} not found in active CANN installation")
    endif()
endforeach()
target_link_libraries(smoke_add_test PRIVATE "${ASCENDCL_LIBRARY}" "${CUST_OPAPI_LIBRARY}" "${ACL_OP_COMPILER_LIBRARY}" "${NNOPBASE_LIBRARY}")
