cmake_minimum_required(VERSION 3.16)
project(raw_probe_host_local LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CANN_ROOT OR NOT HOST_SOURCE OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
 message(FATAL_ERROR "Explicit local CANN Linux ARM64 paths required")
endif()
foreach(name tiling_api register c_sec platform ascendcl)
 find_library(LIB_${name} NAMES ${name} PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
 if(NOT LIB_${name})
  message(FATAL_ERROR "Missing selected CANN library ${name}")
 endif()
endforeach()
add_library(raw_host_local SHARED "${HOST_SOURCE}/d_flash_full_k_int32_probe.cpp")
target_include_directories(raw_host_local PRIVATE "${CANN_ROOT}/include" "${CANN_ROOT}/include/experiment" "${CANN_ROOT}/aarch64-linux/asc/include" "${CANN_ROOT}/aarch64-linux/asc/include/utils" "${CANN_ROOT}/aarch64-linux/asc/include/adv_api")
target_link_libraries(raw_host_local PRIVATE "${LIB_tiling_api}" "${LIB_register}" "${LIB_c_sec}" "${LIB_platform}" "${LIB_ascendcl}" ${CMAKE_DL_LIBS})
target_link_options(raw_host_local PRIVATE -Wl,--no-undefined)
