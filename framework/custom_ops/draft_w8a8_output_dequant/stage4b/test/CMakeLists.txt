cmake_minimum_required(VERSION 3.16)
project(stage4b_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Requires Linux aarch64 CANN receiver")
endif()
foreach(name CANN_ROOT OPP_ROOT COMMON_INCLUDE)
  if(NOT ${name})
    message(FATAL_ERROR "Missing ${name}")
  endif()
endforeach()
set(api "${OPP_ROOT}/vendors/customize/op_api")
foreach(name ascendcl nnopbase acl_op_compiler)
  find_library(LIB_${name} NAMES ${name} PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
  if(NOT LIB_${name})
    message(FATAL_ERROR "Missing ${name}")
  endif()
endforeach()
find_library(LIB_custom NAMES cust_opapi PATHS "${api}/lib" NO_DEFAULT_PATH)
if(NOT LIB_custom)
  message(FATAL_ERROR "Missing independent OPP library")
endif()
add_executable(stage4b_runner runner.cpp)
target_include_directories(stage4b_runner PRIVATE "${CANN_ROOT}/include" "${api}/include" "${COMMON_INCLUDE}")
target_compile_options(stage4b_runner PRIVATE -O2)
target_link_libraries(stage4b_runner PRIVATE "${LIB_custom}" "${LIB_nnopbase}" "${LIB_acl_op_compiler}" "${LIB_ascendcl}")
