cmake_minimum_required(VERSION 3.16)
project(stage4i_host_only LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
 message(FATAL_ERROR "Requires Linux aarch64 receiver; no cross/device Kernel compiler")
endif()
if(NOT CANN_ROOT)
 message(FATAL_ERROR "CANN_ROOT required")
endif()
foreach(name tiling_api register c_sec platform ascendcl)
 find_library(LIB_${name} NAMES ${name} PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
 if(NOT LIB_${name})
  message(FATAL_ERROR "Missing selected CANN library: ${name}")
 endif()
endforeach()
add_executable(stage4i_host probe.cpp)
target_include_directories(stage4i_host PRIVATE "${CANN_ROOT}/include" "${CANN_ROOT}/include/experiment" "${CANN_ROOT}/aarch64-linux/include" "${CANN_ROOT}/aarch64-linux/include/experiment" "${CANN_ROOT}/aarch64-linux/asc/include" "${CANN_ROOT}/aarch64-linux/asc/include/utils" "${CANN_ROOT}/aarch64-linux/asc/include/adv_api")
target_compile_options(stage4i_host PRIVATE -O2)
target_link_libraries(stage4i_host PRIVATE "${LIB_tiling_api}" "${LIB_register}" "${LIB_c_sec}" "${LIB_platform}" "${LIB_ascendcl}" ${CMAKE_DL_LIBS})
