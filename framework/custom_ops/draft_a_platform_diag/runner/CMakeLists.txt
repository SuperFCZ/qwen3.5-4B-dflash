cmake_minimum_required(VERSION 3.16)
project(platform_diag_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(platform_diag_runner main.cpp)
target_include_directories(platform_diag_runner PRIVATE "${CANN_ROOT}/include" "${COMMON_INCLUDE}" "${PRIVATE_OPP}/vendors/customize/op_api/include")
foreach(lib ascendcl nnopbase acl_op_compiler)
 find_library(LIB_${lib} ${lib} PATHS "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64" NO_DEFAULT_PATH)
 if(NOT LIB_${lib})
  message(FATAL_ERROR "Missing CANN library ${lib}")
 endif()
 target_link_libraries(platform_diag_runner PRIVATE "${LIB_${lib}}")
endforeach()
find_library(LIB_custom cust_opapi PATHS "${PRIVATE_OPP}/vendors/customize/op_api/lib" NO_DEFAULT_PATH)
if(NOT LIB_custom)
 message(FATAL_ERROR "Missing private ACLNN library")
endif()
target_link_libraries(platform_diag_runner PRIVATE "${LIB_custom}")
