cmake_minimum_required(VERSION 3.16)
project(dflash_gate_simulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(name CANN_ROOT SIM_LIB OPP_ROOT RUNNER_SOURCE)
    if(NOT ${name})
        message(FATAL_ERROR "Missing ${name}")
    endif()
endforeach()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Simulator receiver must be Linux")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "This diagnostic targets the verified aarch64 simulator installation")
endif()
find_library(SIM_RUNTIME NAMES runtime_camodel PATHS "${SIM_LIB}" NO_DEFAULT_PATH)
find_library(SIM_DRIVER NAMES npu_drv_camodel PATHS "${SIM_LIB}" NO_DEFAULT_PATH)
set(vendor "${OPP_ROOT}/vendors/customize/op_api")
set(sdk_libs "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64")
foreach(name ascendcl acl_op_compiler nnopbase)
    find_library(LIB_${name} NAMES ${name} PATHS ${sdk_libs} NO_DEFAULT_PATH)
endforeach()
find_library(LIB_cust_opapi NAMES cust_opapi PATHS "${vendor}/lib" NO_DEFAULT_PATH)
foreach(name SIM_RUNTIME SIM_DRIVER LIB_ascendcl LIB_acl_op_compiler LIB_nnopbase LIB_cust_opapi)
    if(NOT ${name})
        message(FATAL_ERROR "Missing required simulator/ACLNN library ${name}; no real-driver fallback")
    endif()
endforeach()
add_executable(dflash_gate_simulator "${RUNNER_SOURCE}")
target_include_directories(dflash_gate_simulator PRIVATE "${CANN_ROOT}/include" "${vendor}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(dflash_gate_simulator PRIVATE DFLASH_GROUP_QUANT_K_TILE=256)
target_compile_options(dflash_gate_simulator PRIVATE -g -O2)
# Force model libraries into DT_NEEDED; LD_PRELOAD additionally interposes the
# transitive runtime dependencies used by AscendCL/ACLNN. No real driver lookup.
target_link_libraries(dflash_gate_simulator PRIVATE "-Wl,--no-as-needed" "${SIM_RUNTIME}" "${SIM_DRIVER}"
    "${LIB_ascendcl}" "${LIB_cust_opapi}" "${LIB_acl_op_compiler}" "${LIB_nnopbase}" dl)
set_target_properties(dflash_gate_simulator PROPERTIES BUILD_RPATH "${SIM_LIB};${vendor}/lib;${CANN_ROOT}/lib64")
