cmake_minimum_required(VERSION 3.16)
project(dflash_gate_fp16_tiled_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(name CANN_ROOT OPP_ROOT RUNNER_SOURCE COMMON_INCLUDE)
  if(NOT ${name})
    message(FATAL_ERROR "Missing ${name}")
  endif()
endforeach()
set(api "${OPP_ROOT}/vendors/customize/op_api")
set(paths "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64")
foreach(name ascendcl nnopbase acl_op_compiler)
  find_library(LIB_${name} NAMES ${name} PATHS ${paths} NO_DEFAULT_PATH)
  if(NOT LIB_${name})
    message(FATAL_ERROR "Missing ${name}")
  endif()
endforeach()
find_library(LIB_custom NAMES cust_opapi PATHS "${api}/lib" NO_DEFAULT_PATH)
if(NOT LIB_custom OR NOT EXISTS "${api}/include/aclnn_d_flash_gate_fp16_tiled.h")
  message(FATAL_ERROR "Custom FP16 ACLNN library/header missing")
endif()
add_executable(dflash_gate_fp16_tiled_test "${RUNNER_SOURCE}")
target_include_directories(dflash_gate_fp16_tiled_test PRIVATE "${CANN_ROOT}/include" "${api}/include" "${COMMON_INCLUDE}")
target_compile_definitions(dflash_gate_fp16_tiled_test PRIVATE DFLASH_NATIVE_MM_HEADER="aclnn_d_flash_gate_fp16_tiled.h")
target_compile_options(dflash_gate_fp16_tiled_test PRIVATE -O2)
target_link_libraries(dflash_gate_fp16_tiled_test PRIVATE "${LIB_custom}" "${LIB_nnopbase}" "${LIB_acl_op_compiler}" "${LIB_ascendcl}")
