cmake_minimum_required(VERSION 3.16)
project(fp16_direct_mm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(name CANN_ROOT NATIVE_MM_HEADER COMMON_INCLUDE)
  if(NOT ${name})
    message(FATAL_ERROR "Missing ${name}")
  endif()
endforeach()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
  message(FATAL_ERROR "Requires Linux aarch64 Ascend310P3 receiver")
endif()
set(paths "${CANN_ROOT}/lib64" "${CANN_ROOT}/aarch64-linux/lib64")
foreach(name ascendcl opapi nnopbase)
  find_library(LIB_${name} NAMES ${name} PATHS ${paths} NO_DEFAULT_PATH)
  if(NOT LIB_${name})
    message(FATAL_ERROR "Missing native ${name}; no custom library fallback")
  endif()
endforeach()
add_executable(fp16_direct_mm native_mm.cpp)
target_include_directories(fp16_direct_mm PRIVATE "${CANN_ROOT}/include" "${COMMON_INCLUDE}")
target_compile_definitions(fp16_direct_mm PRIVATE DFLASH_NATIVE_MM_HEADER="${NATIVE_MM_HEADER}")
target_compile_options(fp16_direct_mm PRIVATE -O2)
target_link_libraries(fp16_direct_mm PRIVATE "${LIB_opapi}" "${LIB_nnopbase}" "${LIB_ascendcl}")
